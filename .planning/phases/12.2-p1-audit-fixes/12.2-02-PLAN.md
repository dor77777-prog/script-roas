       // Pre-fix: `start.getTime() + 24h` ms-arithmetic drifted across IL
       // DST transitions (IDT⇄IST, late March + late October) — a backfill

       // Component arithmetic is timezone-agnostic and matches the IL

       // Evidence: raw-returns/inngest_eventBackfill.json (INN-14).
       export function* dateRange(from: string, to: string): Generator<string> {
         const [fy, fm, fd] = from.split('-').map(Number);
         const [ty, tm, td] = to.split('-').map(Number);
         let y = fy, m = fm, d = fd;
         while (y < ty || (y === ty && m < tm) || (y === ty && m === tm && d <= td)) {
           const yyyy = String(y).padStart(4, '0');
           const mm = String(m).padStart(2, '0');
           const dd = String(d).padStart(2, '0');
           yield `${yyyy}-${mm}-${dd}`;
           // Increment by 1 calendar day (component-based).
           d += 1;
           const daysInMonth = new Date(Date.UTC(y, m, 0)).getUTCDate(); // last day of month m
           if (d > daysInMonth) {

             m += 1;
             if (m > 12) {
               m = 1;
               y += 1;
             }
           }
         }
       }
