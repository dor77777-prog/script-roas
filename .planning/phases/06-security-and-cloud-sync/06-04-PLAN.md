```javascript
// ============================================================================
// Phase 6 — dashboard-state-audit retention
//
// The audit tab `dashboard-state-audit` is populated by /api/dashboard-state
// (Phase 6 plan 03) — one row per POST. Without retention, it grows
// unboundedly. This section provides:
//   - pruneAuditLogOlderThan(days): pure pruner, callable manually or via
//     trigger. Returns row counts for visibility.
//   - pruneAuditLogTrigger():     trigger entry point — reads
//     `audit.retention.days` script property (default 30) and delegates.
//   - installAuditPruneTrigger(): one-shot installer to register the weekly
//     time-based trigger. Idempotent: re-running removes the existing
//     trigger first.
//
// Apps Script is standalone (NOT container-bound), so there's no warm
// spreadsheet menu. Run `installAuditPruneTrigger` directly from the
// editor's function dropdown after Phase 6 plan 03 ships.
// ============================================================================

/** Default retention window if script property `audit.retention.days` is unset. */
const AUDIT_RETENTION_DAYS_DEFAULT = 30;
const AUDIT_TAB_NAME = 'dashboard-state-audit';

/**
 * Prunes rows from `dashboard-state-audit` whose column-A timestamp is older
 * than (now - days). Returns { deletedRows, keptRows } for visibility.
 *
 * Edge cases:
 *   - Tab missing (Phase 6 plan 03 not deployed yet OR no POSTs happened):
 *     logs "tab not present" and returns { deletedRows: 0, keptRows: 0 }.
 *   - Tab exists but has only the header row (no data): returns
 *     { deletedRows: 0, keptRows: 0 } without writes.
 *   - `days` is invalid (non-finite, <= 0): logs warning and falls back
 *     to AUDIT_RETENTION_DAYS_DEFAULT to prevent accidental total wipe.
 *
 * Implementation mirrors `archiveTabRows_` (line 743): read all rows,
 * partition to toKeep vs toMove (here we just discard toMove), then
 * rewrite the data range with toKeep only. O(N) in tab size.
 */
function pruneAuditLogOlderThan(days) {
  // Guard against misconfiguration that would wipe everything.
  if (!Number.isFinite(days) || days <= 0) {
    Logger.log(
      'pruneAuditLogOlderThan: invalid days=' + days +
      ', falling back to default ' + AUDIT_RETENTION_DAYS_DEFAULT
    );
    days = AUDIT_RETENTION_DAYS_DEFAULT;
  }

  const ss = ensureSpreadsheet();
  const sheet = ss.getSheetByName(AUDIT_TAB_NAME);
  if (!sheet) {
    Logger.log('pruneAuditLogOlderThan: tab "' + AUDIT_TAB_NAME + '" not present, skipping');
    return { deletedRows: 0, keptRows: 0 };
  }

  const lastRow = sheet.getLastRow();
  if (lastRow <= 1) {
    Logger.log('pruneAuditLogOlderThan: tab empty, nothing to prune');
    return { deletedRows: 0, keptRows: 0 };
  }

  const lastCol = sheet.getLastColumn();
  if (lastCol < 4) {
    Logger.log(
      'pruneAuditLogOlderThan: unexpected column count ' + lastCol +
      ' (expected >= 4) — skipping to avoid corrupting unknown layout'
    );
    return { deletedRows: 0, keptRows: 0 };
  }

  // ISO-8601 strings compare lexicographically, so a cutoff in the same
  // format is the right boundary. We use UTC for the cutoff because the
  // Next.js route writes `new Date().toISOString()` which is also UTC.
  const cutoffMs = Date.now() - days * 24 * 60 * 60 * 1000;
  const cutoffIso = new Date(cutoffMs).toISOString();

  const dataRange = sheet.getRange(2, 1, lastRow - 1, lastCol);
  const rows = dataRange.getValues();

  const toKeep = [];
  let deletedRows = 0;
  for (const r of rows) {
    const ts = r[0];
    // Accept both string (most likely — route writes ISO string) and Date
    // (defensive — operator might paste a Date manually).
    let key = '';
    if (typeof ts === 'string') {
      key = ts;
    } else if (ts instanceof Date && !isNaN(ts.getTime())) {
      key = ts.toISOString();
    }
    // Empty timestamps preserved — could be a manual edit / scratch row.
    // Better to keep unknowns than to silently destroy something.
    if (!key) {
      toKeep.push(r);
      continue;
    }
    if (key < cutoffIso) {
      deletedRows++;
    } else {
      toKeep.push(r);
    }
  }

  Logger.log(
    'pruneAuditLogOlderThan: cutoff=' + cutoffIso +
    ', total=' + rows.length +
    ', to-delete=' + deletedRows +
    ', to-keep=' + toKeep.length
  );

  if (deletedRows === 0) {
    return { deletedRows: 0, keptRows: rows.length };
  }

  // Rewrite the data region with only the kept rows. Pattern adapted from
  // archiveTabRows_ (lines 808–814): clearContent + setValues, NOT
  // deleteRows. deleteRows shifts row indices which can race with any
  // in-flight reads / writes; rewriting keeps the tab geometry stable
  // (just empty trailing rows) which the audit-append logic already handles.
  dataRange.clearContent();
  if (toKeep.length > 0) {
    sheet.getRange(2, 1, toKeep.length, lastCol).setValues(toKeep);
  }

  Logger.log(
    'pruneAuditLogOlderThan: rewrote tab — ' +
    deletedRows + ' rows deleted, ' + toKeep.length + ' rows kept'
  );
  return { deletedRows: deletedRows, keptRows: toKeep.length };
}

/**
 * Weekly trigger entry point. Reads `audit.retention.days` from Script
 * Properties (default AUDIT_RETENTION_DAYS_DEFAULT if unset or invalid).
 * Delegates to pruneAuditLogOlderThan and surfaces the result in logs so
 * a daily report scan can pick it up.
 */
function pruneAuditLogTrigger() {
  const raw = getProp('audit.retention.days', '');
  let days = AUDIT_RETENTION_DAYS_DEFAULT;
  if (raw) {
    const parsed = Number(raw);
    if (Number.isFinite(parsed) && parsed > 0) {
      days = parsed;
    } else {
      Logger.log(
        'pruneAuditLogTrigger: invalid script property audit.retention.days="' + raw +
        '", falling back to default ' + AUDIT_RETENTION_DAYS_DEFAULT
      );
    }
  }
  Logger.log('pruneAuditLogTrigger: running with retention=' + days + ' days');
  try {
    const result = pruneAuditLogOlderThan(days);
    Logger.log('pruneAuditLogTrigger: done — deleted=' + result.deletedRows + ', kept=' + result.keptRows);
  } catch (e) {
    const msg = e && e.message ? e.message : String(e);
    Logger.log('pruneAuditLogTrigger: ERROR — ' + msg);
    // Re-throw so the trigger shows as failed in the Executions tab.
    // Quiet failures hide bugs; a failed execution is visible.
    throw e;
  }
}

/**
 * One-shot installer for the weekly trigger. Idempotent: removes any existing
 * `pruneAuditLogTrigger` registration first, then creates a fresh weekly
 * trigger (Sundays around 03:00 Asia/Jerusalem).
 *
 * Run this manually from the Apps Script editor after Phase 6 plan 03 is
 * deployed (so the audit tab exists or will exist on next POST). Sunday at
 * 03:00 is intentionally low-traffic to avoid colliding with the per-store
 * daily triggers (00:05/08/11) and the 15-min live triggers.
 */
function installAuditPruneTrigger() {
  const HANDLER_NAME = 'pruneAuditLogTrigger';
  const triggers = ScriptApp.getProjectTriggers();
  for (let i = 0; i < triggers.length; i++) {
    if (triggers[i].getHandlerFunction() === HANDLER_NAME) {
      ScriptApp.deleteTrigger(triggers[i]);
    }
  }
  ScriptApp.newTrigger(HANDLER_NAME)
    .timeBased()
    .everyWeeks(1)
    .onWeekDay(ScriptApp.WeekDay.SUNDAY)
    .atHour(3)
    .create();
  Logger.log(
    'Installed ' + HANDLER_NAME + ' weekly (Sunday ~03:00). ' +
    'Retention is ' + AUDIT_RETENTION_DAYS_DEFAULT + ' days by default; ' +
    'override via script property "audit.retention.days".'
  );
}
```

**הסברים נקודתיים:**

1. **למה לא `deleteRows` אלא `clearContent + setValues`?**
   - `deleteRows` משנה row indices, מה שיכול לסכן race עם POST שמנסה להוסיף שורה באותו רגע (לדוגמה: append API).
   - `clearContent + setValues` שומר את ה-tab geometry: שורות ריקות בסוף, אבל ה-`ensureAuditTab_` ב-route.ts תמיד מוסיף שורה בסוף (append), לא לפי row index. אז יש לנו "trailing empty rows" שהקריאה הבאה ל-`spreadsheets.values.append` תדלג עליהן (Sheets API מתעדכן אוטומטית בעבור append).
   - גם: זה ה-pattern של `archiveTabRows_` הקיים — תהיה consistency.

2. **למה Sunday 03:00 ולא midnight?**
   - הdaily triggers ב-Phase 5 רצים ב-00:05/08/11. live triggers ב-15 דקות. Sunday 03:00 רחוק מכל זה. כל כשל בtrigger נצפה ב-Executions tab ביום ראשון בבוקר.

3. **למה לא חושף שום thing בrouting layer (Next.js)?**
   - Single-user dashboard. ה-trigger רץ אוטומטית מ-Apps Script עם service-account credentials שכבר יש Editor. הוספת route ב-Next.js עם secret-based access הייתה מוסיפה complexity ללא ערך.
   - חלופה (לא בוחרים): operator יכול בעתיד לקרוא ל-`pruneAuditLogOlderThan(60)` ידנית מ-editor אם נדרש pruning חד-פעמי.

4. **למה לסנר `Number.isFinite` ולא רק `> 0`?**
   - `NaN > 0` הוא `false`, אבל אם property הוא `"foo"` (string לא-numeric), `Number("foo") = NaN`. `Number.isFinite(NaN) = false`. שני הבדיקות יחד מבטיחות שאיכשהו `0` חוקי לא יעבור (אם operator יכתוב `0` במכוון — fallback ל-default).

5. **למה לזרוק ב-trigger handler במקום לבלוע?**
   - Apps Script Executions tab מראה success/failure לכל trigger run. אם נבלע שגיאה, ניראה הצלחה גם אם הtab נמחק בטעות. לזרוק = visible failure בUI הניהול.

```markdown
## 🛡️ אבטחה — Phase 6 (Slim scope, מאי 2026)

הדשבורד הוא כלי **פנימי לסיווג משתמש-יחיד**: לכל היותר משתמש אחד עורך באותו זמן, אין surface DDoS, ו-URL obscurity הוא גבול האמון. Phase 6 הוקטן בכוונה ל-3 דרישות שמשתלבות במודל האמון הזה ולא דורשות auth retrofit:

### 1. Service-account split (06-01)

שני service accounts במקום אחד, כדי להגביל את ה-blast radius של דליפת credential:

| SA | OAuth Scope | משמש את | משתני סביבה ב-Vercel |
|----|-------------|---------|----------------------|
| Reader | `spreadsheets.readonly` | כל ה-GET endpoints (`/api/data`, `/api/campaigns`, `/api/products`, `/api/orders-attribution`, GET `/api/dashboard-state`, `/api/store-meta`) | `GOOGLE_READER_EMAIL`, `GOOGLE_READER_KEY` |

### 2. Audit log (06-03)

טאב חדש `dashboard-state-audit` בspreadsheet עם 4 עמודות:

```
A = timestamp (ISO-8601)
B = key (אחד מ-ALLOWED_STATE_KEYS)
C = old_value (JSON-stringified, truncated 500 chars)
D = new_value (JSON-stringified, truncated 500 chars)
```

כל POST שמצליח ב-`/api/dashboard-state` מוסיף שורה אחת לטאב **לפני** שהוא מחזיר 200 לקליינט. ה-append הוא best-effort — אם Sheets API נכשל, ה-write הראשי עדיין מצליח, ה-audit miss נרשם ב-`console.warn`, וה-append הבא ימלא שורה. נוצר אוטומטית ב-POST הראשון (`ensureAuditTab_` יוצר את ה-tab + header row).

**Retention**: 30 ימים, ע"י Apps Script trigger שבועי (`pruneAuditLogTrigger`, יום ראשון ~03:00). ה-trigger מתקין דרך `installAuditPruneTrigger()` שמופעל ידנית פעם אחת מה-editor של Apps Script (פרויקט standalone — אין onOpen menu). ניתן לשנות retention דרך Script Property `audit.retention.days`.

### 3. Adaptive polling (06-02)

הdashboard ה-`CloudSync` היה polls ל-`/api/dashboard-state` כל 30 שניות, גם כשהטאב מוסתר וגם כשהמשתמש כבר לא ראה את המסך זמן רב. השדרוג:

| מצב | תדר polling |
|-----|-------------|
| Tab visible | 30s |
| Tab hidden | 5min |
| Idle > 10min (אין mousemove/keydown/touchstart) | עצור עד activity הבא |

זה חוסך ~10x מהקריאות הברקע (טאב שכוח-פתוח לילה שלם = 0 קריאות במקום 120 בשעה), משאיר את ה-Sheets read quota לדפלויים עתידיים שיכניסו עוד routes או שותפים.

### מה לא בPhase 6 ולמה (slim scope)

| נדחה | סיבה |
|------|------|
| Rate limiting (Upstash Redis) | משתמש יחיד, אין שטח DDoS |
| If-Match optimistic concurrency | משתמש יחיד = אין concurrent edits = אין race |
| Auth layer (basic auth / Cloudflare Access) | המשתמש קיבל URL obscurity כמודל אמון |

אם דרישת multi-user תתעורר בעתיד, השדרוגים האלה יכולים להיכנס ב-phase נפרד מבלי לסתור את מה שPhase 6 בנה — ה-split של SA וה-audit log כבר מספקים את התשתית.
```
