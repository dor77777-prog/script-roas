
    Should become:
    ```js

    const dayEnd   = isoLocalMidnight_(nextDayStr_(dateStr));
    ```

    Where `isoLocalMidnight_(dateStr)` is a NEW helper that returns a TZ-aware ISO string for local midnight on `dateStr`. Place the helper at the top of Shopify.gs (after the existing constants).
  </behavior>
  <action>
    1. Open `Shopify.gs`.
    2. At the top (after existing constants, before the first function), add the helper:
       ```javascript
       /**
        * Returns ISO-8601 timestamp for local midnight on `dateStr` (YYYY-MM-DD)
        * in the project timezone (TZ = Asia/Jerusalem). Automatically resolves
        * the correct UTC offset (+02:00 winter, +03:00 DST) — replaces the
        * hardcoded `+03:00` that silently corrupted winter day windows.
        *
        * Fix for CODEX-NEW-P0-01.
        */
       function isoLocalMidnight_(dateStr) {
         if (!/^\d{4}-\d{2}-\d{2}$/.test(dateStr)) {
           throw new Error(`isoLocalMidnight_: invalid dateStr "${dateStr}"`);
         }
         // Construct a Date at local midnight on the target day in TZ. Using
         // Utilities.formatDate with the XXX pattern resolves the offset.
         const [y, m, d] = dateStr.split('-').map(Number);
         // Local midnight in TZ is what we want; Utilities.formatDate against a
         // UTC Date adjusted to TZ-local midnight is the canonical way:
         const localMidnight = new Date(Date.UTC(y, m - 1, d, 0, 0, 0));
         // The trick: format that "UTC point" in TZ, then ALSO format its actual
         // offset. We just need the TZ offset for `dateStr` at midnight.
         // Simpler: pick the target instant as 12:00 UTC to safely avoid DST
         // transition windows, format it in TZ to get the date, then reconstruct
         // midnight + offset.
         const noonUtc = new Date(Date.UTC(y, m - 1, d, 12, 0, 0));
         const offsetStr = Utilities.formatDate(noonUtc, TZ, 'XXX'); // → '+02:00' or '+03:00'
         return `${dateStr}T00:00:00${offsetStr}`;
       }
       ```
