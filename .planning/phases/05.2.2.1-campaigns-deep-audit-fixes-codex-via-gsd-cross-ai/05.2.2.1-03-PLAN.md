       export function computeDayDelta(channelTotal: number, shopify: number): { label: string; tone: 'neutral' | 'green' | 'red' } {
         if (channelTotal === 0 && shopify > 0) return { label: 'Shopify only', tone: 'neutral' };
         if (channelTotal > 0 && shopify === 0) return { label: 'Channels only', tone: 'neutral' };
         if (channelTotal > 0 && shopify > 0) {

           const label = `${pct >= 0 ? '+' : ''}${pct.toFixed(0)}%`;
           const tone: 'green' | 'red' = Math.abs(pct) < 20 ? 'green' : 'red';
           return { label, tone };
