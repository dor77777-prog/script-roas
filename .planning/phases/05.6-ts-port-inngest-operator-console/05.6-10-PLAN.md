    ```typescript
    import { inngest } from '@/inngest/client';
    import { runDailyForStore } from './cronDaily';

    type SyncNowPayload = {
      storeId: 'uzoshop' | 'zolplus' | 'usmile360';
      date?: string;  // YYYY-MM-DD; defaults to today in Asia/Jerusalem

    function todayJerusalem(): string {
      const fmt = new Intl.DateTimeFormat('en-CA', {
        timeZone: 'Asia/Jerusalem',
        year: 'numeric', month: '2-digit', day: '2-digit',
      });
      return fmt.format(new Date());
    }

      { event: 'event/sync-now' },
      async ({ event, step }) => {
        const { storeId, date = todayJerusalem() } = event.data as SyncNowPayload;
        return runDailyForStore(storeId, date, { step });
      },
    );
    ```

    ```typescript
    // dashboard-web/src/inngest/functions/eventBackfill.ts
    import { inngest } from '@/inngest/client';
    import { runDailyForStore } from './cronDaily';

    type StoreId = 'uzoshop' | 'zolplus' | 'usmile360';
    type BackfillPayload = {
      from: string;       // YYYY-MM-DD inclusive
      to: string;         // YYYY-MM-DD inclusive
      storeIds: StoreId[];
    };

    const HISTORY_BOUNDARY = '2026-05-01';  // D-A3 — no data exists before this date

    function* dateRange(from: string, to: string): Generator<string> {
      const [fy, fm, fd] = from.split('-').map(Number);
      const [ty, tm, td] = to.split('-').map(Number);
      const start = new Date(Date.UTC(fy, fm - 1, fd));
      const end = new Date(Date.UTC(ty, tm - 1, td));
      for (let d = start; d <= end; d = new Date(d.getTime() + 24 * 60 * 60 * 1000)) {
        yield d.toISOString().slice(0, 10);
      }
    }

    export const eventBackfill = inngest.createFunction(
      { id: 'event-backfill' },
      { event: 'event/backfill' },
      async ({ event, step }) => {
        const { from, to, storeIds } = event.data as BackfillPayload;

        // Validation
        if (!storeIds || storeIds.length === 0) {
          throw new Error('event/backfill: storeIds array is required and must be non-empty');
        }
        if (from < HISTORY_BOUNDARY) {
          throw new Error(
            `event/backfill: from=${from} predates D-A3 history boundary ${HISTORY_BOUNDARY}; ` +
            `Phase 05.6 only supports backfill from 2026-05-01 forward.`,
          );
        }
        if (from > to) {
          throw new Error(`event/backfill: from=${from} > to=${to}`);
        }

        const results: Array<{ date: string; storeId: string; ok: boolean; error?: string }> = [];

        for (const date of dateRange(from, to)) {
          for (const storeId of storeIds) {
            try {
              await step.run(`backfill-${date}-${storeId}`, () =>
                runDailyForStore(storeId, date, { step }),
              );
              results.push({ date, storeId, ok: true });
            } catch (err) {
              const message = err instanceof Error ? err.message : String(err);
              results.push({ date, storeId, ok: false, error: message });
              // Continue to next (date, store) on failure — let Inngest's per-step retry
              // mechanic handle the transient case; if it dead-letters, we keep marching
              // and the operator sees the failed step in the jobs table.
            }
          }
        }

        return {
          from, to, storeIds,
          totalSteps: results.length,
          successCount: results.filter(r => r.ok).length,
          failureCount: results.filter(r => !r.ok).length,
        };
      },
    );
    ```

    Note: each `step.run('backfill-{date}-{storeId}', ...)` itself wraps a call to `runDailyForStore`, which internally makes another ~5 step.runs. This means a backfill of 3 days × 3 stores generates ~9 outer steps × ~5 inner steps + 9 outer + 1 function = ~55 execs. For a 21-day backfill (D-A3 full history from 2026-05-01 to 2026-05-21) × 3 stores = ~315 outer + 1575 inner + 1 = ~1900 execs — single one-shot. Free tier (50K/month) absorbs this 25x over.

    The nested step.run pattern is documented as supported by Inngest (functions can call other step.run via the same `step` instance passed through the handler context). If `runDailyForStore` were instantiated as a separate Inngest function and we used `step.invoke()`, that would be a future refinement — for 05.6, direct call within the same handler is simpler.
