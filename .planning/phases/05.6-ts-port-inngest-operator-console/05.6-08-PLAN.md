    ```typescript
    // dashboard-web/src/inngest/functions/cronDaily.ts
    import { inngest } from '@/inngest/client';
    import { fetchShopifyDayRows } from '@/lib/fetchers/shopify';
    import { fetchMetaAdSetInsights, fetchMetaSpendForDay } from '@/lib/fetchers/meta';
    import { fetchGoogleAdsSpendForDay, fetchGoogleAdsAdGroupInsights } from '@/lib/fetchers/googleAds';
    import { getFxRate } from '@/lib/fetchers/fx';
    import { mergeOverridesFromSupabase } from '@/lib/fetchers/manualOverrides';
    import { getSupabaseAdmin } from '@/lib/supabaseAdmin';

    const STORES = ['uzoshop', 'zolplus', 'usmile360'] as const;
    type StoreId = typeof STORES[number];

    // Matches Apps Script Config.gs:78-82 (yesterdayStr_) — yesterday in Asia/Jerusalem.
    function yesterdayJerusalem(): string {
      const fmt = new Intl.DateTimeFormat('en-CA', {
        timeZone: 'Asia/Jerusalem',
        year: 'numeric', month: '2-digit', day: '2-digit',
      });
      const now = new Date(Date.now() - 24 * 60 * 60 * 1000);
      return fmt.format(now); // YYYY-MM-DD
    }

    // Shared handler body — also called by event/sync-now (plan 10) and event/backfill (plan 10).
    export async function runDailyForStore(
      storeId: StoreId,
      dateStr: string,
      ctx: { step: any /* Inngest StepTools — fully typed by Inngest's createFunction signature */ },
    ) {
      const { step } = ctx;

      // 1. fetch-shopify
      const shopify = await step.run('fetch-shopify', () => fetchShopifyDayRows(storeId, dateStr));

      // 2. fetch-meta (spend + adset/ad-level insights)
      const meta = await step.run('fetch-meta', async () => {
        const [spend, adsetRows] = await Promise.all([
          fetchMetaSpendForDay(storeId, dateStr),
          fetchMetaAdSetInsights(storeId, dateStr),
        ]);
        return { spend, adsetRows };
      });

      // 3. fetch-google (uzoshop only — others short-circuit inside the fetcher)
      const google = await step.run('fetch-google', async () => {
        const [spend, adGroupRows] = await Promise.all([
          fetchGoogleAdsSpendForDay(storeId, dateStr),
          fetchGoogleAdsAdGroupInsights(storeId, dateStr),
        ]);
        return { spend, adGroupRows };
      });

      // 4. apply-manual-overrides (returns CAD-normalized fb+ga+total)
      const merged = await step.run('apply-manual-overrides', () =>
        mergeOverridesFromSupabase({
          storeId, date: dateStr,
          metaSpend: meta.spend,    // {spend, currency:'ILS'}
          googleSpend: google.spend, // {spend, currency:'CAD'} or {spend:0, currency:'CAD'}
        }),
      );

      // 5. persist-batch — all 4-5 tables in one step (Pitfall 4: keep steps low)
      await step.run('persist-batch', async () => {
        const admin = getSupabaseAdmin();
        const roas = merged.totalSpendCad > 0 ? shopify.revenueCad / merged.totalSpendCad : 0;
        const grossProfit = shopify.revenueCad - merged.totalSpendCad;
        const cogs = shopify.revenueCad * 0.25; // COGS_RATE_OF_REVENUE per Config.gs:20
        const netProfit = grossProfit - cogs;

        // data_daily UPSERT (PK: date, store_id)
        let { error } = await admin.from('data_daily').upsert({
          date: dateStr, store_id: storeId, store_name: shopify.storeName,
          fb_spend_cad: merged.fbSpendCad,
          ga_spend_cad: merged.gaSpendCad,
          total_spend_cad: merged.totalSpendCad,
          revenue_cad: shopify.revenueCad,
          roas, gross_profit_cad: grossProfit, cogs_cad: cogs, net_profit_cad: netProfit,
        }, { onConflict: 'date,store_id' });
        if (error) throw new Error(`data_daily upsert: ${error.message}`);

        // products_daily UPSERT (PK: date, store_id, product_id)
        if (shopify.productRows.length > 0) {
          ({ error } = await admin.from('products_daily').upsert(
            shopify.productRows.map(p => ({
              date: dateStr, store_id: storeId, store_name: shopify.storeName,
              product_id: p.product_id,
              net_revenue_cad: p.net_revenue_cad,
              // gross_revenue, units, orders, product_title — out of scope for cronDaily;
              // populated by a future per-product fetcher or left null
            })),
            { onConflict: 'date,store_id,product_id' },
          ));
          if (error) throw new Error(`products_daily upsert: ${error.message}`);
        }

        // campaigns_daily UPSERT (PK: date, store_id, platform, campaign_id, ad_set_id)
        const campaignRows = meta.adsetRows.map(r => ({
          date: dateStr, store_id: storeId, platform: 'meta',
          campaign_id: r.campaignId, campaign_name: r.campaignName,
          ad_set_id: r.adSetId, ad_set_name: r.adSetName,
          // spend_cad: needs ILS→CAD conversion — use the same FX rate as Meta spend conversion
          // For simplicity, leave NULL when Meta-only and conversion not pre-computed; plan 09 (cron-live) refines
          spend_cad: null, impressions: r.impressions, clicks: r.clicks,
          conversions: r.conversions, conversion_value_cad: null, roas: null,
          campaign_budget_cad: null, ad_set_budget_cad: null, budget_type: null,
        }));
        if (campaignRows.length > 0) {
          ({ error } = await admin.from('campaigns_daily').upsert(campaignRows, {
            onConflict: 'date,store_id,platform,campaign_id,ad_set_id',
          }));
          if (error) throw new Error(`campaigns_daily upsert: ${error.message}`);
        }

        // ads_daily UPSERT (PK: date, store_id, ad_id) — only if google.adGroupRows non-empty
        // (Note: ads_daily is for ad-level rows, not ad-group — schema needs verification.
        // The current schema's PK on ad_id suggests Meta-style ad-level data, NOT Google ad-group.
        // For Google Ads ad-groups, they map closer to campaigns_daily.ad_set_id semantics.
        // We persist Google ad-group rows into campaigns_daily as well, with platform='google'.)
        const googleCampaignRows = google.adGroupRows.map(r => ({
          date: dateStr, store_id: storeId, platform: 'google',
          campaign_id: r.campaignId, campaign_name: r.campaignName,
          ad_set_id: r.adGroupId, ad_set_name: r.adGroupName,
          spend_cad: r.spend, // Google Ads returns CAD already for uzoshop
          impressions: r.impressions, clicks: r.clicks,
          conversions: r.conversions, conversion_value_cad: r.conversionValue, roas: null,
          campaign_budget_cad: null, ad_set_budget_cad: null, budget_type: null,
        }));
        if (googleCampaignRows.length > 0) {
          ({ error } = await admin.from('campaigns_daily').upsert(googleCampaignRows, {
            onConflict: 'date,store_id,platform,campaign_id,ad_set_id',
          }));
          if (error) throw new Error(`campaigns_daily (google) upsert: ${error.message}`);
        }
      });

      return {
        storeId, date: dateStr,
        shopifyRevenueCad: shopify.revenueCad,
        fbSpendCad: merged.fbSpendCad,
        gaSpendCad: merged.gaSpendCad,
        totalSpendCad: merged.totalSpendCad,
        overridesApplied: merged.overridesApplied,
        productRowCount: shopify.productRows.length,
        metaCampaignRowCount: meta.adsetRows.length,
        googleCampaignRowCount: google.adGroupRows.length,
      };
    }

    // Factory: one inngest.createFunction per store
    function makeCronDaily(storeId: StoreId) {
      return inngest.createFunction(
        { id: `cron-daily-${storeId}` },
        { cron: 'TZ=Asia/Jerusalem 5 0 * * *' },
        async ({ step }) => {
          const date = yesterdayJerusalem();
          return runDailyForStore(storeId, date, { step });
        },
      );
    }

    export const cronDailyFunctions = STORES.map(makeCronDaily);
    ```

    **Step budget**: 5 step.run calls per function + 1 function invocation = 6 execs/run. × 3 stores × 1 run/day = 18 execs/day = ~540/month. WELL under 50K cap.

    **Step granularity rationale**: per RESEARCH §Pitfall 4, batching 5 table writes into ONE persist step is the recommended decomposition. A transient failure re-runs all 5 writes (idempotent via ON CONFLICT — no double-deduction risk).
