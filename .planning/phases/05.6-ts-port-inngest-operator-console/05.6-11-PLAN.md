    ```typescript
    // dashboard-web/src/app/api/inngest/route.ts
    //
    // Inngest webhook endpoint. Receives POST from Inngest cloud for every function
    // invocation. The serve() helper returns {GET, POST, PUT} — Next.js app router
    // auto-wires these.
    //
    //   PUT: Inngest calls during a Vercel deploy to register / update functions.
    //   GET: returns function metadata + (dev only) a landing page.
    //   POST: actual function invocation entry — signed with INNGEST_SIGNING_KEY.
    //
    // INNGEST_SIGNING_KEY auto-read from process.env at request time.
    //
    // maxDuration: 60s per RESEARCH §Pitfall 9. Vercel Hobby caps at 10s — if the
    // operator is on Hobby, large-day Shopify fetches (50 pages × ~500ms each)
    // may exceed the cap. Pro plan (60s) is required for production reliability.
    // If on Hobby, recommend operator upgrade OR decompose Shopify pagination into
    // smaller step.runs in plans 03 / 08.

    import { serve } from 'inngest/next';
    import { inngest } from '@/inngest/client';
    import { cronDailyFunctions } from '@/inngest/functions/cronDaily';
    import { cronLiveFunctions } from '@/inngest/functions/cronLive';
    import { eventSyncNow } from '@/inngest/functions/eventSyncNow';
    import { eventBackfill } from '@/inngest/functions/eventBackfill';

    // Vercel route config:
    //   - maxDuration: 60s (Pro plan; if Hobby this is silently ignored at 10s)
    //   - NO `force-dynamic` (RESEARCH §Pitfall 11 — conflicts with revalidate; we want neither here)
    //   - NO `revalidate` either (this route is invoked by Inngest, not cached)
    //
    // Per Next.js route segment config: maxDuration must be a literal number.
    export const maxDuration = 60;

    export const { GET, POST, PUT } = serve({
      client: inngest,
      functions: [
        ...cronDailyFunctions,   // 3 functions (uzoshop / zolplus / usmile360)
        ...cronLiveFunctions,    // 3 functions
        eventSyncNow,            // 1 function
        eventBackfill,           // 1 function
      ],
      // INNGEST_SIGNING_KEY auto-read from process.env at runtime.
    });
    ```

    Notes:
    - DO NOT use `force-dynamic` (Pitfall 11)
    - DO NOT use `revalidate` (this is an invocation endpoint, not a cacheable GET)
    - `maxDuration = 60` is the Vercel Pro plan max — operator confirms plan in plan 01 Task 0 SUMMARY
