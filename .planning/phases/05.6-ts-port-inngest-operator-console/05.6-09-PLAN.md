<objective>
Create 3 Inngest cron-live functions (one per store) running every 15 minutes in Asia/Jerusalem. Each refreshes Shopify rolling 3-day data ONLY (not Meta/Google — see budget rationale below), then persists `data_daily` + `products_daily`. Matches Apps Script's live trigger semantics minus the 6-min execution cap.

Purpose: Live cadence keeps the dashboard near-real-time for revenue (Shopify is the variable signal — orders + refunds change throughout the day). Meta/Google insights only update hourly anyway, so polling them every 15min wastes budget. The 3-day rolling window catches cross-day refunds within a 72h horizon (matches Apps Script's `rollingBackfillDays = 3`).

Output: 1 functions file + 1 test file. Free-tier budget: ~3 execs/run × 3 stores × 96 runs/day = ~864 execs/day = ~26K/month. Combined with cron-daily (~540/month) total ~27K/month = 54% of 50K cap (RESEARCH §Pitfall 4 reconciliation).
</objective>

    ```typescript
    // dashboard-web/src/inngest/functions/cronLive.ts
    import { inngest } from '@/inngest/client';
    import { fetchShopifyDayRows } from '@/lib/fetchers/shopify';
    import { getSupabaseAdmin } from '@/lib/supabaseAdmin';

    const STORES = ['uzoshop', 'zolplus', 'usmile360'] as const;
    type StoreId = typeof STORES[number];

    function rollingWindowJerusalem(daysBack: number = 3): string[] {
      const fmt = new Intl.DateTimeFormat('en-CA', {
        timeZone: 'Asia/Jerusalem',
        year: 'numeric', month: '2-digit', day: '2-digit',
      });
      const dates: string[] = [];
      const today = new Date();
      for (let i = 0; i < daysBack; i++) {
        const d = new Date(today.getTime() - i * 24 * 60 * 60 * 1000);
        dates.push(fmt.format(d));
      }
      return dates;  // [today, today-1, today-2]
    }

    export async function runLiveForStore(
      storeId: StoreId,
      ctx: { step: any },
    ) {
      const { step } = ctx;
      const dates = rollingWindowJerusalem(3);

      // Single fetch step: fetches all 3 days in parallel for the store.
      const shopifyByDate = await step.run('fetch-shopify-rolling-3day', async () => {
        const results = await Promise.all(
          dates.map(d => fetchShopifyDayRows(storeId, d)),
        );
        return Object.fromEntries(dates.map((d, i) => [d, results[i]]));
      });

      // Single persist step: UPSERTs data_daily + products_daily for each of 3 dates.
      await step.run('persist-rolling-3day', async () => {
        const admin = getSupabaseAdmin();
        for (const date of dates) {
          const shopify = shopifyByDate[date];
          // Note: data_daily.fb_spend_cad / ga_spend_cad / total_spend_cad are NOT updated
          // by the live trigger (the daily cron owns those). We use UPDATE-only-revenue
          // by reading the existing row first.
          //
          // Actually — the simpler path: read existing row, merge revenue update, upsert back.
          // OR: use a partial update via .update().match() instead of upsert.
          // Choosing .update() per RESEARCH §Architectural Map row 3 — preserves the daily-cron's
          // spend values without overwriting them to NULL.

          const { error: updateErr } = await admin.from('data_daily').update({
            store_name: shopify.storeName,
            revenue_cad: shopify.revenueCad,
            // Note: roas, gross_profit, cogs, net_profit need recomputation against existing
            // spend values. We fetch the existing row first, compute, then update.
          }).eq('date', date).eq('store_id', storeId);

          // If no row exists yet (e.g., the daily cron hasn't run for today), insert a partial row.
          // SELECT first to decide INSERT-or-UPDATE:
          if (updateErr) throw new Error(`data_daily live update: ${updateErr.message}`);

          const { data: existing, error: selErr } = await admin
            .from('data_daily')
            .select('id, fb_spend_cad, ga_spend_cad, total_spend_cad')
            .eq('date', date).eq('store_id', storeId)
            .maybeSingle();
          if (selErr) throw new Error(`data_daily select: ${selErr.message}`);

          if (!existing) {
            // No daily row yet — insert with revenue only (spend will be filled by cron-daily).
            const { error: insErr } = await admin.from('data_daily').upsert({
              date, store_id: storeId, store_name: shopify.storeName,
              revenue_cad: shopify.revenueCad,
              fb_spend_cad: 0, ga_spend_cad: 0, total_spend_cad: 0,
              roas: 0, gross_profit_cad: shopify.revenueCad,
              cogs_cad: shopify.revenueCad * 0.25,
              net_profit_cad: shopify.revenueCad - shopify.revenueCad * 0.25,
            }, { onConflict: 'date,store_id' });
            if (insErr) throw new Error(`data_daily live insert: ${insErr.message}`);
          } else {
            // Existing row — recompute derived columns against persisted spend.
            const totalSpend = Number(existing.total_spend_cad) || 0;
            const roas = totalSpend > 0 ? shopify.revenueCad / totalSpend : 0;
            const grossProfit = shopify.revenueCad - totalSpend;
            const cogs = shopify.revenueCad * 0.25;
            const netProfit = grossProfit - cogs;
            const { error: upErr } = await admin.from('data_daily').update({
              revenue_cad: shopify.revenueCad,
              roas, gross_profit_cad: grossProfit,
              cogs_cad: cogs, net_profit_cad: netProfit,
            }).eq('date', date).eq('store_id', storeId);
            if (upErr) throw new Error(`data_daily live derived: ${upErr.message}`);
          }

          // products_daily — upsert net_revenue_cad per product
          if (shopify.productRows.length > 0) {
            const { error: pErr } = await admin.from('products_daily').upsert(
              shopify.productRows.map(p => ({
                date, store_id: storeId, store_name: shopify.storeName,
                product_id: p.product_id, net_revenue_cad: p.net_revenue_cad,
              })),
              { onConflict: 'date,store_id,product_id' },
            );
            if (pErr) throw new Error(`products_daily live upsert: ${pErr.message}`);
          }
        }
      });

      return {
        storeId,
        rollingDates: dates,
        rowsTouched: dates.length * 2, // data_daily + products_daily per date
      };
    }

    function makeCronLive(storeId: StoreId) {
      return inngest.createFunction(
        { id: `cron-live-${storeId}` },
        { cron: 'TZ=Asia/Jerusalem */15 * * * *' },
        async ({ step }) => runLiveForStore(storeId, { step }),
      );
    }

    export const cronLiveFunctions = STORES.map(makeCronLive);
    ```

    **Step budget**: 2 step.runs + 1 function = 3 execs/run. × 3 stores × 96 runs/day × 30 days = 25,920/month. Plus cron-daily ~540/month. Total ~26,460/month = **53% of free tier**. Plenty of headroom.

    **Rationale for skipping Meta/Google in live**: Meta insights for "today" are stale (Meta aggregates hourly); refreshing every 15min vs hourly costs 4x the API calls for the same data. Apps Script's existing live trigger also only refreshes Shopify (per `DailyUpdate.gs` behavior). Algorithm parity (D-C4) demands the same.
