<objective>
Port `ManualOverrides.gs` (11KB) lookup logic to TS, READING from Supabase `manual_overrides` table (NOT the Sheets `manual-spend` tab). This is THE load-bearing piece per D-A5 — without manual_overrides, uzoshop's May 1-8 spend rows are blank (Meta's API will never return the original disabled ad account's data).

Purpose: The merge function applies operator-typed overrides AFTER Meta/Google fetchers return spend values but BEFORE the persist step. If `(date, store_id, platform='meta')` has a row in manual_overrides, the Meta-fetched spend for that day/store is replaced by the row's value (converted to CAD via fx.ts if currency != CAD).

Output: 1 module with the merge function + tests. The Sheets-edit helpers from `ManualOverrides.gs` (lines that WRITE to Sheets) are NOT ported — Plan 15's CRUD UI handles writes.
</objective>

    ```typescript
    // dashboard-web/src/lib/fetchers/manualOverrides.ts
    //
    // Reads manual_overrides table from Supabase and merges into per-day spend values
    // before persist. Replaces (does not add to) the fetched spend for any platform
    // that has an override row — mirrors ManualOverrides.gs:71-99.
    //
    // Uses getSupabaseAdmin() because the table grants SELECT only to anon — but
    // reading via service_role keeps us on the same client used by all writers,
    // simplifying the trust boundary surface.

    import { getSupabaseAdmin } from '@/lib/supabaseAdmin';
    import { getFxRate } from '@/lib/fetchers/fx';

    export type SpendInput = { spend: number; currency: string };

    export type MergeInput = {
      storeId: string;
      date: string;             // YYYY-MM-DD
      metaSpend: SpendInput;
      googleSpend: SpendInput;
    };

    export type MergeResult = {
      fbSpendCad: number;
      gaSpendCad: number;
      totalSpendCad: number;
      overridesApplied: { meta: boolean; google: boolean };  // for SUMMARY + logging
    };

    type OverrideRow = {
      date: string;
      store_id: string;
      platform: 'meta' | 'google';
      spend: number | string;   // NUMERIC(14, 4) — supabase-js may return as string
      currency: string;
    };

    async function spendToCad(spend: SpendInput, dateStr: string): Promise<number> {
      if (spend.currency === 'CAD') return spend.spend;
      const rate = await getFxRate(spend.currency, 'CAD', dateStr);
      return spend.spend * rate;
    }

    async function overrideToCad(row: OverrideRow, dateStr: string): Promise<number> {
      const amount = typeof row.spend === 'string' ? parseFloat(row.spend) : row.spend;
      if (!Number.isFinite(amount)) {
        throw new Error(`manual_overrides spend not numeric for ${row.store_id} ${row.date} ${row.platform}`);
      }
      if (row.currency === 'CAD') return amount;
      const rate = await getFxRate(row.currency, 'CAD', dateStr);
      return amount * rate;
    }

    export async function mergeOverridesFromSupabase(input: MergeInput): Promise<MergeResult> {
      const { storeId, date, metaSpend, googleSpend } = input;

      const { data, error } = await getSupabaseAdmin()
        .from('manual_overrides')
        .select('date,store_id,platform,spend,currency')
        .eq('store_id', storeId)
        .eq('date', date);

      if (error) {
        throw new Error(`manual_overrides lookup failed for ${storeId} ${date}: ${error.message}`);
      }

      const rows = (data ?? []) as OverrideRow[];
      let fbSpendCad: number;
      let gaSpendCad: number;
      const overridesApplied = { meta: false, google: false };

      const metaRow = rows.find(r => r.platform === 'meta');
      if (metaRow) {
        fbSpendCad = await overrideToCad(metaRow, date);
        overridesApplied.meta = true;
      } else {
        fbSpendCad = await spendToCad(metaSpend, date);
      }

      const googleRow = rows.find(r => r.platform === 'google');
      if (googleRow) {
        gaSpendCad = await overrideToCad(googleRow, date);
        overridesApplied.google = true;
      } else {
        gaSpendCad = await spendToCad(googleSpend, date);
      }

      // Defensive: rows with platform other than meta/google are caught by the
      // CHECK constraint at write-time, so this branch should be unreachable.
      // Log if it ever happens (would indicate a constraint bypass / corruption).
      for (const r of rows) {
        if (r.platform !== 'meta' && r.platform !== 'google') {
          console.warn(`manual_overrides row with unexpected platform '${r.platform}' for ${storeId} ${date} — skipped`);
        }
      }

      const totalSpendCad = fbSpendCad + gaSpendCad;
      return { fbSpendCad, gaSpendCad, totalSpendCad, overridesApplied };
    }
    ```

    Note: this design returns CAD values for ALL three numbers, replacing the Apps Script approach of returning native currency + a flag. The Inngest function (plan 08) writes these directly to `data_daily.fb_spend_cad`, `data_daily.ga_spend_cad`, `data_daily.total_spend_cad` — schema matches.
