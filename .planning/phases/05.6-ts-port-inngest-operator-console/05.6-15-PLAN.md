    ```typescript
    import { NextResponse } from 'next/server';
    import { getSupabaseAdmin } from '@/lib/supabaseAdmin';
    import { userFacingError } from '@/lib/apiErrors';

    export const dynamic = 'force-dynamic';

    const VALID_STORES = new Set(['uzoshop', 'zolplus', 'usmile360']);
    const VALID_PLATFORMS = new Set(['meta', 'google']);
    const VALID_CURRENCIES = new Set(['ILS', 'CAD', 'USD']);

    type PostBody = {
      date: string; store_id: string; platform: string;
      spend: number; currency: string; notes?: string | null;
    };

    function isDate(v: unknown): v is string {
      return typeof v === 'string' && /^\d{4}-\d{2}-\d{2}$/.test(v);
    }

    function validatePost(body: any): PostBody | string {
      if (!isDate(body.date)) return 'date must be YYYY-MM-DD';
      if (!VALID_STORES.has(body.store_id)) return `unknown store_id: ${body.store_id}`;
      const platform = String(body.platform || '').toLowerCase();
      if (!VALID_PLATFORMS.has(platform)) return `platform must be 'meta' or 'google'`;
      const currency = String(body.currency || '').toUpperCase();
      if (!VALID_CURRENCIES.has(currency)) return `currency must be ILS/CAD/USD`;
      const spend = parseFloat(String(body.spend));
      if (!Number.isFinite(spend)) return 'spend must be numeric';
      return { date: body.date, store_id: body.store_id, platform, spend, currency, notes: body.notes ?? null };
    }

    export async function GET() {
      try {
        const { data, error } = await getSupabaseAdmin()
          .from('manual_overrides')
          .select('id, date, store_id, platform, spend, currency, notes, created_at')
          .order('date', { ascending: false })
          .order('store_id', { ascending: true });
        if (error) {
          return NextResponse.json({ rows: [], error: userFacingError(error.message) }, { status: 200 });
        }
        return NextResponse.json({ rows: data ?? [] });
      } catch (err) {
        const message = err instanceof Error ? err.message : String(err);
        return NextResponse.json({ rows: [], error: userFacingError(message) }, { status: 200 });
      }
    }

    export async function POST(req: Request) {
      try {
        const raw = await req.json();
        const valid = validatePost(raw);
        if (typeof valid === 'string') {
          return NextResponse.json({ error: valid }, { status: 400 });
        }
        const { data, error } = await getSupabaseAdmin()
          .from('manual_overrides')
          .upsert(valid, { onConflict: 'date,store_id,platform' })
          .select();
        if (error) return NextResponse.json({ error: userFacingError(error.message) }, { status: 400 });
        return NextResponse.json({ row: data?.[0] }, { status: 201 });
      } catch (err) {
        const message = err instanceof Error ? err.message : String(err);
        console.error('/api/operator/manual-overrides POST failed:', message);
        return NextResponse.json({ error: userFacingError(message) }, { status: 500 });
      }
    }

    export async function PATCH(req: Request) {
      try {
        const body = await req.json() as { id: number } & Partial<PostBody>;
        if (!body.id || typeof body.id !== 'number') {
          return NextResponse.json({ error: 'id required' }, { status: 400 });
        }
        const { id, ...patch } = body;
        // Normalize platform if present
        if (patch.platform) patch.platform = String(patch.platform).toLowerCase();
        if (patch.currency) patch.currency = String(patch.currency).toUpperCase();
        const { data, error } = await getSupabaseAdmin()
          .from('manual_overrides')
          .update(patch)
          .eq('id', id)
          .select();
        if (error) return NextResponse.json({ error: userFacingError(error.message) }, { status: 400 });
        return NextResponse.json({ row: data?.[0] });
      } catch (err) {
        const message = err instanceof Error ? err.message : String(err);
        console.error('/api/operator/manual-overrides PATCH failed:', message);
        return NextResponse.json({ error: userFacingError(message) }, { status: 500 });
      }
    }

    export async function DELETE(req: Request) {
      try {
        const body = await req.json() as { id: number };
        if (!body.id || typeof body.id !== 'number') {
          return NextResponse.json({ error: 'id required' }, { status: 400 });
        }
        const { error } = await getSupabaseAdmin()
          .from('manual_overrides')
          .delete()
          .eq('id', body.id);
        if (error) return NextResponse.json({ error: userFacingError(error.message) }, { status: 400 });
        return NextResponse.json({ ok: true });
      } catch (err) {
        const message = err instanceof Error ? err.message : String(err);
        console.error('/api/operator/manual-overrides DELETE failed:', message);
        return NextResponse.json({ error: userFacingError(message) }, { status: 500 });
      }
    }
    ```

    ```typescript
    'use client';

    import useSWR from 'swr';
    import { useState } from 'react';
    import { Plus, Trash2, X } from 'lucide-react';

    type Row = {
      id: number;
      date: string;
      store_id: string;
      platform: string;
      spend: number;
      currency: string;
      notes: string | null;
    };

    const ALL_STORES = ['uzoshop', 'zolplus', 'usmile360'] as const;
    const ALL_PLATFORMS = ['meta', 'google'] as const;
    const ALL_CURRENCIES = ['ILS', 'CAD', 'USD'] as const;

    async function fetcher(url: string): Promise<{ rows: Row[]; error?: string }> {
      const res = await fetch(url);
      return (await res.json()) as { rows: Row[]; error?: string };
    }

    export function ManualOverridesCrud() {
      const { data, error, mutate, isLoading } = useSWR(
        '/api/operator/manual-overrides',
        fetcher,
      );
      const [confirmDelete, setConfirmDelete] = useState<Row | null>(null);
      const [submitting, setSubmitting] = useState(false);
      const [formState, setFormState] = useState({
        date: new Date().toISOString().slice(0, 10),
        store_id: 'uzoshop',
        platform: 'meta',
        spend: 0,
        currency: 'ILS',
        notes: '',
      });
      const [addError, setAddError] = useState<string | null>(null);

      const addRow = async () => {
        setSubmitting(true);
        setAddError(null);
        try {
          const res = await fetch('/api/operator/manual-overrides', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({
              ...formState,
              platform: formState.platform.toLowerCase(),
              currency: formState.currency.toUpperCase(),
              notes: formState.notes || null,
            }),
          });
          const body = await res.json();
          if (res.status >= 400) throw new Error(body.error ?? `HTTP ${res.status}`);
          await mutate();  // refresh table
          setFormState(s => ({ ...s, spend: 0, notes: '' }));
        } catch (e) {
          setAddError(e instanceof Error ? e.message : String(e));
        } finally {
          setSubmitting(false);
        }
      };

      const deleteRow = async (id: number) => {
        try {
          const res = await fetch('/api/operator/manual-overrides', {
            method: 'DELETE',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ id }),
          });
          if (res.status >= 400) {
            const body = await res.json();
            throw new Error(body.error ?? `HTTP ${res.status}`);
          }
          await mutate();
          setConfirmDelete(null);
        } catch (e) {
          // Surface in confirm modal text
          setConfirmDelete(prev => prev ? { ...prev, notes: `שגיאה: ${e instanceof Error ? e.message : String(e)}` } : null);
        }
      };

      if (isLoading) return <p className="text-text-secondary text-sm">טוען…</p>;
      if (error) return <p className="text-red-400 text-sm">שגיאה: {String(error)}</p>;
      if (data?.error) return <p className="text-amber-400 text-sm">{data.error}</p>;

      return (
        <div className="space-y-3">
          {/* Add row form */}
          <div className="flex flex-wrap items-end gap-2 p-3 bg-black/20 rounded">
            <input type="date" value={formState.date} onChange={e => setFormState(s => ({ ...s, date: e.target.value }))}
              className="bg-black/30 border border-white/10 rounded px-2 py-1 text-sm" dir="ltr" />
            <select value={formState.store_id} onChange={e => setFormState(s => ({ ...s, store_id: e.target.value }))}
              className="bg-black/30 border border-white/10 rounded px-2 py-1 text-sm">
              {ALL_STORES.map(s => <option key={s} value={s}>{s}</option>)}
            </select>
            <select value={formState.platform} onChange={e => setFormState(s => ({ ...s, platform: e.target.value }))}
              className="bg-black/30 border border-white/10 rounded px-2 py-1 text-sm">
              {ALL_PLATFORMS.map(p => <option key={p} value={p}>{p}</option>)}
            </select>
            <input type="number" step="0.01" value={formState.spend}
              onChange={e => setFormState(s => ({ ...s, spend: parseFloat(e.target.value) || 0 }))}
              className="bg-black/30 border border-white/10 rounded px-2 py-1 text-sm w-24" dir="ltr" placeholder="spend" />
            <select value={formState.currency} onChange={e => setFormState(s => ({ ...s, currency: e.target.value }))}
              className="bg-black/30 border border-white/10 rounded px-2 py-1 text-sm">
              {ALL_CURRENCIES.map(c => <option key={c} value={c}>{c}</option>)}
            </select>
            <input type="text" value={formState.notes} onChange={e => setFormState(s => ({ ...s, notes: e.target.value }))}
              className="bg-black/30 border border-white/10 rounded px-2 py-1 text-sm flex-1 min-w-40" placeholder="הערות (אופציונלי)" />
            <button onClick={addRow} disabled={submitting} className="flex items-center gap-1 bg-blue-600 hover:bg-blue-500 disabled:bg-gray-600 text-white text-sm px-3 py-1 rounded">
              <Plus className="w-4 h-4" /> הוסף
            </button>
          </div>
          {addError && <p className="text-red-400 text-sm">{addError}</p>}

          {/* Table */}
          <div className="overflow-x-auto">
            <table className="min-w-full text-sm">
              <thead className="text-text-secondary text-xs uppercase tracking-wider">
                <tr>
                  <th className="text-right p-2">תאריך</th>
                  <th className="text-right p-2">חנות</th>
                  <th className="text-right p-2">פלטפורמה</th>
                  <th className="text-right p-2">סכום</th>
                  <th className="text-right p-2">מטבע</th>
                  <th className="text-right p-2">הערות</th>
                  <th className="text-right p-2">פעולות</th>
                </tr>
              </thead>
              <tbody>
                {(data?.rows ?? []).map(r => (
                  <tr key={r.id} className="border-t border-white/10">
                    <td className="p-2" dir="ltr">{r.date}</td>
                    <td className="p-2">{r.store_id}</td>
                    <td className="p-2">{r.platform}</td>
                    <td className="p-2" dir="ltr">{Number(r.spend).toFixed(2)}</td>
                    <td className="p-2" dir="ltr">{r.currency}</td>
                    <td className="p-2 text-text-secondary">{r.notes ?? '—'}</td>
                    <td className="p-2">
                      <button onClick={() => setConfirmDelete(r)} className="text-red-400 hover:text-red-300">
                        <Trash2 className="w-4 h-4" />
                      </button>
                    </td>
                  </tr>
                ))}
              </tbody>
            </table>
          </div>

          {/* Delete confirmation modal */}
          {confirmDelete && (
            <div className="fixed inset-0 z-50 flex items-center justify-center bg-black/60">
              <div className="bg-background border border-white/10 rounded p-4 max-w-md">
                <div className="flex items-start justify-between mb-3">
                  <h3 className="text-lg font-semibold">אישור מחיקה</h3>
                  <button onClick={() => setConfirmDelete(null)}><X className="w-4 h-4" /></button>
                </div>
                <p className="text-sm mb-3">
                  למחוק את ההחלפה הידנית של {confirmDelete.store_id} בתאריך {confirmDelete.date} (פלטפורמה {confirmDelete.platform})?
                </p>
                {confirmDelete.notes?.startsWith('שגיאה:') && (
                  <p className="text-red-400 text-sm mb-3">{confirmDelete.notes}</p>
                )}
                <div className="flex justify-end gap-2">
                  <button onClick={() => setConfirmDelete(null)} className="bg-gray-600 hover:bg-gray-500 text-white text-sm px-3 py-1 rounded">ביטול</button>
                  <button onClick={() => deleteRow(confirmDelete.id)} className="bg-red-600 hover:bg-red-500 text-white text-sm px-3 py-1 rounded">מחק</button>
                </div>
              </div>
            </div>
          )}

          <p className="text-text-secondary text-xs">
            סה״כ {(data?.rows ?? []).length} שורות. שינויים נכנסים לתוקף בריצת ה-Inngest הבאה (cron-daily, sync-now, או backfill).
          </p>
        </div>
      );
    }
    ```
