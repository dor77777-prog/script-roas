    // dashboard-web/src/app/api/operator/sync-now/route.ts
    //
    // Fires event/sync-now for one or all stores. Returns 202 immediately;
    // operator watches /api/operator/jobs for completion (async semantics
    // per D-D4 + RESEARCH §Open Question 4).

    import { NextResponse } from 'next/server';
    import { inngest } from '@/inngest/client';
    import { userFacingError } from '@/lib/apiErrors';

    export const dynamic = 'force-dynamic';

    type Payload =
      | { scope: 'all' }
      | { scope: 'store'; storeId: 'uzoshop' | 'zolplus' | 'usmile360' };

    const ALL_STORES = ['uzoshop', 'zolplus', 'usmile360'] as const;
    const VALID_STORES = new Set(ALL_STORES);

    export async function POST(req: Request) {
      try {
        const body = await req.json() as Payload;

        let events: Array<{ name: 'event/sync-now'; data: { storeId: string } }>;
        if (body.scope === 'all') {
          events = ALL_STORES.map(s => ({ name: 'event/sync-now', data: { storeId: s } }));
        } else if (body.scope === 'store' && VALID_STORES.has(body.storeId)) {
          events = [{ name: 'event/sync-now', data: { storeId: body.storeId } }];
        } else {
          return NextResponse.json(
            { error: `Invalid payload: scope must be 'all' or 'store' with valid storeId` },
            { status: 400 },
          );
        }

        const result = await inngest.send(events);
        return NextResponse.json(
          { accepted: result.ids.length, eventIds: result.ids },
          { status: 202 },
        );
      } catch (err) {
        const message = err instanceof Error ? err.message : String(err);
        console.error('/api/operator/sync-now POST failed:', message);
        return NextResponse.json({ error: userFacingError(message) }, { status: 500 });
      }
    }
    ```

    // dashboard-web/src/app/api/operator/backfill/route.ts

    import { NextResponse } from 'next/server';
    import { inngest } from '@/inngest/client';
    import { userFacingError } from '@/lib/apiErrors';

    export const dynamic = 'force-dynamic';

    const ALL_STORES = ['uzoshop', 'zolplus', 'usmile360'] as const;
    const VALID_STORES = new Set(ALL_STORES);
    const HISTORY_BOUNDARY = '2026-05-01'; // D-A3 (matches plan 10's eventBackfill validation)

    type Payload = {
      from: string;       // YYYY-MM-DD
      to: string;
      storeIds: string[];
    };

    function isDate(s: unknown): s is string {
      return typeof s === 'string' && /^\d{4}-\d{2}-\d{2}$/.test(s);
    }

    export async function POST(req: Request) {
      try {
        const body = await req.json() as Payload;

        if (!isDate(body.from) || !isDate(body.to)) {
          return NextResponse.json(
            { error: 'from and to must be YYYY-MM-DD strings' },
            { status: 400 },
          );
        }
        if (body.from < HISTORY_BOUNDARY) {
          return NextResponse.json(
            { error: `from=${body.from} predates D-A3 history boundary ${HISTORY_BOUNDARY}` },
            { status: 400 },
          );
        }
        if (body.from > body.to) {
          return NextResponse.json(
            { error: `from=${body.from} > to=${body.to}` },
            { status: 400 },
          );
        }
        if (!Array.isArray(body.storeIds) || body.storeIds.length === 0) {
          return NextResponse.json(
            { error: 'storeIds must be a non-empty array' },
            { status: 400 },
          );
        }
        const invalid = body.storeIds.find(s => !VALID_STORES.has(s as any));
        if (invalid) {
          return NextResponse.json(
            { error: `Invalid storeId: ${invalid}` },
            { status: 400 },
          );
        }

        const result = await inngest.send({
          name: 'event/backfill',
          data: { from: body.from, to: body.to, storeIds: body.storeIds },
        });

        return NextResponse.json(
          { accepted: result.ids.length, eventIds: result.ids, range: { from: body.from, to: body.to }, storeIds: body.storeIds },
          { status: 202 },
        );
      } catch (err) {
        const message = err instanceof Error ? err.message : String(err);
        console.error('/api/operator/backfill POST failed:', message);
        return NextResponse.json({ error: userFacingError(message) }, { status: 500 });
      }
    }
    ```

    Step 2 — `BackfillPicker.tsx`:

    ```typescript
    'use client';

    import { useState } from 'react';
    import { CalendarDays, Loader2 } from 'lucide-react';

    const ALL_STORES = ['uzoshop', 'zolplus', 'usmile360'] as const;
    const HISTORY_BOUNDARY = '2026-05-01';

    export function BackfillPicker() {
      const todayIso = new Date().toISOString().slice(0, 10);
      const [from, setFrom] = useState(HISTORY_BOUNDARY);
      const [to, setTo] = useState(todayIso);
      const [storeIds, setStoreIds] = useState<Set<string>>(new Set(ALL_STORES));
      const [submitting, setSubmitting] = useState(false);
      const [message, setMessage] = useState<string | null>(null);
      const [error, setError] = useState<string | null>(null);

      const submit = async () => {
        setSubmitting(true);
        setMessage(null);
        setError(null);
        try {
          const res = await fetch('/api/operator/backfill', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ from, to, storeIds: Array.from(storeIds) }),
          });
          if (res.status !== 202) {
            const body = await res.json().catch(() => ({}));
            throw new Error(body.error ?? `HTTP ${res.status}`);
          }
          const body = await res.json() as { accepted: number; eventIds: string[] };
          setMessage(`Backfill קיבל ${body.accepted} אירועים. עקוב אחר ההתקדמות בטבלת הריצות.`);
        } catch (e) {
          setError(e instanceof Error ? e.message : String(e));
        } finally {
          setSubmitting(false);
        }
      };

      const toggleStore = (s: string) => {
        const next = new Set(storeIds);
        if (next.has(s)) next.delete(s); else next.add(s);
        setStoreIds(next);
      };

      return (
        <div className="space-y-3">
          <div className="flex flex-wrap items-end gap-3">
            <label className="flex flex-col text-sm">
              <span className="text-text-secondary mb-1">מתאריך</span>
              <input
                type="date"
                value={from}
                min={HISTORY_BOUNDARY}
                onChange={e => setFrom(e.target.value)}
                className="bg-black/30 border border-white/10 rounded px-2 py-1"
                dir="ltr"
              />
            </label>
            <label className="flex flex-col text-sm">
              <span className="text-text-secondary mb-1">עד תאריך</span>
              <input
                type="date"
                value={to}
                min={from}
                onChange={e => setTo(e.target.value)}
                className="bg-black/30 border border-white/10 rounded px-2 py-1"
                dir="ltr"
              />
            </label>
            <div className="flex flex-col text-sm">
              <span className="text-text-secondary mb-1">חנויות</span>
              <div className="flex gap-2">
                {ALL_STORES.map(s => (
                  <label key={s} className="flex items-center gap-1 text-sm">
                    <input
                      type="checkbox"
                      checked={storeIds.has(s)}
                      onChange={() => toggleStore(s)}
                    />
                    <span>{s}</span>
                  </label>
                ))}
              </div>
            </div>
            <button
              onClick={submit}
              disabled={submitting || storeIds.size === 0 || from < HISTORY_BOUNDARY || from > to}
              className="flex items-center gap-1 bg-blue-600 hover:bg-blue-500 disabled:bg-gray-600 disabled:cursor-not-allowed text-white text-sm px-3 py-1 rounded"
            >
              {submitting ? <Loader2 className="w-4 h-4 animate-spin" /> : <CalendarDays className="w-4 h-4" />}
              הפעל Backfill
            </button>
          </div>
          {message && <p className="text-green-400 text-sm">{message}</p>}
          {error && <p className="text-red-400 text-sm">שגיאה: {error}</p>}
          <p className="text-text-secondary text-xs">
            * Backfill מקצה ~6 step.runs לכל יום-חנות. טווח של 21 ימים × 3 חנויות ≈ 380 ביצועים (פחות מ-1% מתקרת Inngest).
          </p>
        </div>
      );
    }
    ```
