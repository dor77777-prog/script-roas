<objective>
Lock the cross-day refund algorithm behavior with regression tests that the multi-round audit-and-review process can rely on. The rolled-back attempt failed because fictional fixtures encoded the wrong algorithm and no test caught the divergence from Shopify's real payload shape. This plan creates the structural fix: a pure-TS mirror + Vitest tests with fixtures sourced FROM the empirical probe (Plan 01's PROBE-EVIDENCE.md).

Per D-C3, four invariants are mandatory:
1. **Store-level:** `revenue(D) = sum current_total_price(orders.created_at=D, excl test/voided) - sum transactions[].amount(refunds.processed_at=D AND order.created_at != D)`
2. **Per-product:** `sum byProduct[pid].netRevenueCad(D) = (sum per-line gross for orders.created_at=D after intra-order refund deduction) - sum refund_line_items.subtotal(refunds.processed_at=D AND order.created_at != D, grouped by product_id)`
3. **Cross-path documentation:** store-level approximately equals sum of per-product, with a documented gap = `total_tax_refunded(D) + shipping_refunded(D)`. The gap is documented in code comments on the per-product function.
4. **Full reconciliation:** `store_net(D) == sum byProduct[pid].netRevenueCad(D) + total_tax_refunded(D) + shipping_refunded(D)` — exact equality within ±0.01 CAD.

Per D-C4, fixtures live ONLY where derivable from `05.2.3.0-PROBE-EVIDENCE.md`. NO fictional fixtures. Test names cite the D-C3 invariant number. Acceptance criteria grep-gate the binding between fixtures and the probe file.

      total?: string | number;

    /**
     * Compute the local YYYY-MM-DD date string for an ISO timestamp in the given TZ.
     * Mirrors Apps Script's `Utilities.formatDate(new Date(ts), TZ, 'yyyy-MM-dd')`.
     */
    function dayInTz(ts: string, tz: string): string {
      if (!ts) return '';
      const d = new Date(ts);
      if (!Number.isFinite(d.getTime())) return '';
      // Intl.DateTimeFormat with 'en-CA' gives 'YYYY-MM-DD' shape.
      const fmt = new Intl.DateTimeFormat('en-CA', {
        timeZone: tz,
        year: 'numeric',
        month: '2-digit',
        day: '2-digit',
      });
      return fmt.format(d);
    }

    function parseNum(v: unknown): number {
      if (typeof v === 'number') return Number.isFinite(v) ? v : 0;
      if (typeof v === 'string') {
        const n = parseFloat(v);
        return Number.isFinite(n) ? n : 0;
      }
      return 0;
    }

    export function computeRevenueWithCrossDayRefunds(
      orders: ShopifyOrderInput[],
      dateStr: string,
      tz: string = 'Asia/Jerusalem',
    ): CrossDayRefundResult {
      let grossForD = 0;
      let storeRefundCad = 0;
      let sumRefundLineItemsCad = 0;
      const byProduct: Record<string, { netRevenueCad: number }> = Object.create(null);

      for (const o of orders) {
        if (o.test === true) continue;
        if (o.financial_status === 'voided') continue;

        const orderCreatedDay = dayInTz(o.created_at, tz);

        // Gross side: sum current_total_price for orders created on day D.
        if (orderCreatedDay === dateStr) {
          grossForD += parseNum(o.current_total_price);
        }

        // Refund side: cross-day filter — refund.processed_at on D AND order.created_at NOT on D.
        const refunds = o.refunds ?? [];
        for (const r of refunds) {
          if (!r.processed_at) continue;
          const processedDay = dayInTz(r.processed_at, tz);
          if (processedDay !== dateStr) continue;
          if (orderCreatedDay === dateStr) continue;

          // D-C1: store-level — sum transactions.amount (tax-inclusive).
          const txs = r.transactions ?? [];
          let refundTxSum = 0;
          for (const tx of txs) {
            refundTxSum += parseNum(tx.amount);
          }
          storeRefundCad += refundTxSum;

          // D-C2: per-product — sum refund_line_items.subtotal grouped by product_id.
          const rlis = r.refund_line_items ?? [];
          let refundLineItemSum = 0;
          for (const rli of rlis) {
            const pid = rli.product_id == null ? '' : String(rli.product_id);
            if (!pid) continue; // skip custom / deleted-product items
            const amt = parseNum(rli.subtotal !== undefined ? rli.subtotal : (rli.total ?? 0));
            if (!byProduct[pid]) byProduct[pid] = { netRevenueCad: 0 };
            byProduct[pid].netRevenueCad -= amt;
            refundLineItemSum += amt;
          }
          sumRefundLineItemsCad += refundLineItemSum;
        }
      }

      const storeNetCad = grossForD - storeRefundCad;
      // D-C3 invariant 3: the gap between transactions.amount and refund_line_items.subtotal
      // equals tax + shipping refunds. The current Shopify Admin REST payload does not cleanly
      // split tax-refunded from shipping-refunded — we surface the combined gap here as
      // totalTaxRefundedCad and leave shippingRefundedCad = 0 for forward compatibility.
      const totalTaxRefundedCad = storeRefundCad - sumRefundLineItemsCad;
      const shippingRefundedCad = 0;

      return { storeNetCad, byProduct, totalTaxRefundedCad, shippingRefundedCad };
    }
